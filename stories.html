document.addEventListener("DOMContentLoaded", function() { const feedContainer = document.getElementById('substack-feed'); const paginationContainer = document.getElementById('pagination-controls'); const storiesPerPage = 9; let currentPage = 1; // Function to fetch a specific page from the Substack API function fetchPage(page) { // Calculate how many stories to skip based on the current page const offset = (page - 1) * storiesPerPage; // Use limit and offset to only pull 9 stories at a time const substackUrl = `https://norfolkfolkloresociety.substack.com/api/v1/archive?sort=new&limit=${storiesPerPage}&offset=${offset}`; const apiUrl = `https://api.allorigins.win/raw?url=${encodeURIComponent(substackUrl)}`; // Show loading state feedContainer.innerHTML = '

Summoning stories from the archive...

'; paginationContainer.innerHTML = ''; // Clear buttons while loading fetch(apiUrl) .then(response => { if (!response.ok) throw new Error("Network response was not ok"); return response.json(); }) .then(data => { if (Array.isArray(data) && data.length > 0) { renderStories(data); renderPaginationControls(page, data.length); } else if (page === 1) { feedContainer.innerHTML = '

No stories found at this time.

'; } else { feedContainer.innerHTML = '

You have reached the end of the archive.

'; } }) .catch(error => { console.error('Error fetching Substack API:', error); feedContainer.innerHTML = '

There was an error summoning the archive.

'; }); } // Function to render the fetched stories to the grid function renderStories(stories) { feedContainer.innerHTML = ''; stories.forEach(item => { const dateObj = new Date(item.post_date); const formattedDate = dateObj.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' }); const imageUrl = item.cover_image || 'images/Featured%20Stories%201.png'; let textContent = item.description || ""; const excerpt = textContent.length > 120 ? textContent.substring(0, 120).trim() + "..." : textContent; let author = "Norfolk Folklore Society"; if (item.authors && item.authors.length > 0) { author = item.authors[0].name; } const cardHTML = `

${item.title}

by ${author}
${formattedDate}

${excerpt}

 `; feedContainer.insertAdjacentHTML('beforeend', cardHTML); }); // Re-trigger scroll animations for the new cards document.querySelectorAll('.story-card').forEach(element => { element.classList.add('animate-on-scroll'); observer.observe(element); }); } // Function to render Previous/Next buttons function renderPaginationControls(activePage, itemsReturned) { paginationContainer.innerHTML = ''; // Previous Button const prevBtn = document.createElement('button'); prevBtn.innerHTML = '❮ Previous'; prevBtn.disabled = activePage === 1; prevBtn.addEventListener('click', () => { currentPage--; fetchPage(currentPage); scrollToTop(); }); paginationContainer.appendChild(prevBtn); // Next Button (Disable if we received fewer than 9 items, meaning it's the last page) const nextBtn = document.createElement('button'); nextBtn.innerHTML = 'Next ❯'; nextBtn.disabled = itemsReturned < storiesPerPage; nextBtn.addEventListener('click', () => { currentPage++; fetchPage(currentPage); scrollToTop(); }); paginationContainer.appendChild(nextBtn); } function scrollToTop() { document.querySelector('.section-title').scrollIntoView({ behavior: 'smooth' }); } // Scroll Animation Logic Setup const observerOptions = { root: null, rootMargin: '0px', threshold: 0.15 }; const observer = new IntersectionObserver((entries, observer) => { entries.forEach(entry => { if (entry.isIntersecting) { entry.target.classList.add('is-visible'); observer.unobserve(entry.target); } }); }, observerOptions); // Initial load fetchPage(currentPage); });
